cmake_minimum_required(VERSION 3.19)
project(mdo_harmony_probe LANGUAGES C)

# An isolated probe, not an mdo/xs product port. Point XS_SOURCE_DIR at the
# checkout for the exact xserver commit in mdo's deps.lock.
set(XS_SOURCE_DIR "" CACHE PATH "Pinned xs checkout")
if(NOT EXISTS "${XS_SOURCE_DIR}/tcc/libtcc.c")
    message(FATAL_ERROR "XS_SOURCE_DIR must contain the pinned xs tcc sources")
endif()

add_library(probe_tcc STATIC "${XS_SOURCE_DIR}/tcc/libtcc.c")
target_include_directories(probe_tcc PUBLIC "${XS_SOURCE_DIR}/tcc")
# Use TCC's existing embedded-predefines mode. The tiny probe then needs no
# host include directories or resource-file extraction on the target device.
file(READ "${XS_SOURCE_DIR}/tcc/include/tccdefs.h" TCC_PREDEFS_TEXT)
string(REPLACE "\\" "\\\\" TCC_PREDEFS_TEXT "${TCC_PREDEFS_TEXT}")
string(REPLACE "\"" "\\\"" TCC_PREDEFS_TEXT "${TCC_PREDEFS_TEXT}")
string(REPLACE "\r" "" TCC_PREDEFS_TEXT "${TCC_PREDEFS_TEXT}")
string(REPLACE "\n" "\\n\"\n\"" TCC_PREDEFS_TEXT "${TCC_PREDEFS_TEXT}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/tccdefs_.h" "\"${TCC_PREDEFS_TEXT}\"\n")
target_include_directories(probe_tcc PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(probe_tcc PRIVATE CONFIG_TCC_PREDEFS=1)
set_target_properties(probe_tcc PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$"
        OR OHOS_ARCH STREQUAL "arm64-v8a")
    target_compile_definitions(probe_tcc PRIVATE TCC_TARGET_ARM64)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$"
        OR OHOS_ARCH STREQUAL "x86_64")
    target_compile_definitions(probe_tcc PRIVATE TCC_TARGET_X86_64)
else()
    message(FATAL_ERROR "Only x86_64 and ARM64 are within this assessment")
endif()
if(WIN32)
    target_compile_definitions(probe_tcc PRIVATE TCC_TARGET_PE)
    # The xs fork routes Windows paths through its UTF-8 adapter.
    target_sources(probe_tcc PRIVATE "${XS_SOURCE_DIR}/tcc/tcc_utf8_io.c")
endif()

add_executable(tcc_probe tcc_probe.c)
target_compile_features(tcc_probe PRIVATE c_std_11)
target_link_libraries(tcc_probe PRIVATE probe_tcc)
if(NOT WIN32)
    target_link_libraries(tcc_probe PRIVATE dl m pthread)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "OHOS")
    target_link_libraries(tcc_probe PRIVATE hilog_ndk.z)
    add_library(mdoprobe SHARED napi_probe.c tcc_probe.c platform_probe.c)
    target_include_directories(mdoprobe PRIVATE "${XS_SOURCE_DIR}/lib")
    target_compile_features(mdoprobe PRIVATE c_std_11)
    target_compile_definitions(mdoprobe PRIVATE MDO_HARMONY_SHARED_PROBE=1)
    target_link_libraries(mdoprobe PRIVATE probe_tcc ace_napi.z hilog_ndk.z child_process dl m pthread)
    add_library(mdochildprobe SHARED child_probe.c)
    target_compile_features(mdochildprobe PRIVATE c_std_11)
endif()

option(MDO_PROBE_XRT "Compile the current mdo xrt module selection" OFF)
if(MDO_PROBE_XRT)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "OHOS")
        message(FATAL_ERROR "MDO_PROBE_XRT is only intended for the OHOS cross-compile assessment")
    endif()
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/host-profile.json" MDO_XRT_PROFILE)
    string(JSON MDO_XRT_ROOT_COUNT LENGTH "${MDO_XRT_PROFILE}" xrt_modules)
    math(EXPR MDO_XRT_ROOT_LAST "${MDO_XRT_ROOT_COUNT} - 1")
    add_executable(xrt_compile_probe xrt_compile_probe.c)
    target_include_directories(xrt_compile_probe PRIVATE "${XS_SOURCE_DIR}/lib")
    foreach(MDO_XRT_ROOT_INDEX RANGE ${MDO_XRT_ROOT_LAST})
        string(JSON MDO_XRT_ROOT GET "${MDO_XRT_PROFILE}" xrt_modules ${MDO_XRT_ROOT_INDEX})
        target_compile_definitions(xrt_compile_probe PRIVATE "XRT_MODULE_${MDO_XRT_ROOT}")
    endforeach()
    target_link_libraries(xrt_compile_probe PRIVATE dl m pthread)
endif()
